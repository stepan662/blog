{{/* ------------------------------------------------------------------ Sharing metadata: canonical URL, Open Graph, Twitter card. Without these a shared link renders as a bare URL — no title, no blurb, no image — on X, LinkedIn, Slack, Discord, Bluesky, Mastodon and Reddit. Every value falls back down a chain, so a post that sets nothing still gets something sensible: title — page title, then site title descr — front-matter description, then the post's own summary, then the site description image — front-matter `image`, then the site-wide card from tools/og OG image URLs must be absolute, and the file has to be a raster format: no crawler renders SVG. ------------------------------------------------------------------ */}} {{- $descr := "" }} {{- with .Description }}{{ $descr = . }} {{- else }}{{ with .Summary }}{{ $descr = (. | plainify | chomp | truncate 200) }} {{- else }}{{ $descr = site.Params.description }}{{ end }}{{ end }} {{- /* One card for the whole site — the unfurl prints the title beside it, so the image is the mark, not the headline. A post can still opt out with `image:` in its front matter. */}} {{- $image := .Params.image | default "/blog/og/default.png" }} {{ if .IsHome }}{{ site.Title }}{{ else }}{{ .Title }} · {{ site.Title }}{{ end }} {{- /* OG wants language_TERRITORY (en_US), not the BCP-47 en-us Hugo carries. */}} {{- $loc := split site.Language.Locale "-" }} {{- if and .IsPage (not .Date.IsZero) }} {{- if ne .Lastmod .Date }} {{- end }} {{- range .Params.tags }} {{- end }} {{- end }} {{- with site.Params.twitter }} {{- end }}
