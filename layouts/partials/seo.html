{{/* ------------------------------------------------------------------ Sharing metadata: canonical URL, Open Graph, Twitter card. Without these a shared link renders as a bare URL — no title, no blurb, no image — on X, LinkedIn, Slack, Discord, Bluesky, Mastodon and Reddit. Every value falls back down a chain, so a post that sets nothing still gets something sensible: title — page title, then site title descr — front-matter description, then the post's own summary, then the site description image — front-matter `image`, then the card generated for this page by tools/og (see its README), then the site-wide default card OG image URLs must be absolute, and the file has to be a raster format: no crawler renders SVG. ------------------------------------------------------------------ */}} {{- $descr := "" }} {{- with .Description }}{{ $descr = . }} {{- else }}{{ with .Summary }}{{ $descr = (. | plainify | chomp | truncate 200) }} {{- else }}{{ $descr = site.Params.description }}{{ end }}{{ end }} {{- /* Locate the card image. `.File` is nil on generated pages (taxonomies), hence the guard before deriving a per-page path from the source file. */}} {{- $ogDefault := "/blog/og/default.png" }} {{- $image := $ogDefault }} {{- with .Params.image }} {{- $image = . }} {{- else }} {{- with .File }} {{- $candidate := printf "/blog/og/%s.png" .ContentBaseName }} {{- if fileExists (printf "static%s" $candidate) }}{{ $image = $candidate }}{{ end }} {{- end }} {{- end }} {{ if .IsHome }}{{ site.Title }}{{ else }}{{ .Title }} · {{ site.Title }}{{ end }} {{- /* OG wants language_TERRITORY (en_US), not the BCP-47 en-us Hugo carries. */}} {{- $loc := split site.Language.Locale "-" }} {{- if and .IsPage (not .Date.IsZero) }} {{- if ne .Lastmod .Date }} {{- end }} {{- range .Params.tags }} {{- end }} {{- end }} {{- with site.Params.twitter }} {{- end }}
