{{/* ------------------------------------------------------------------ Sharing metadata: canonical URL, Open Graph, Twitter card. Without these a shared link renders as a bare URL — no title, no blurb, no image — on X, LinkedIn, Slack, Discord, Bluesky, Mastodon and Reddit. Every value falls back down a chain, so a post that sets nothing still gets something sensible: title — page title, then site title descr — front-matter description, then the post's own summary, then the site description ogdesc — front-matter ogDescription, then descr. These are split on purpose: Google's snippet rewards 120-160 characters, while a social preview shows roughly 125 before truncating on mobile. One string can't be ideal for both, so a post that cares can write a tighter one for the card and keep the long one for search. image — front-matter `image`, then the site-wide card from tools/og OG image URLs must be absolute, and the file has to be a raster format: no crawler renders SVG. ------------------------------------------------------------------ */}} {{- $descr := "" }} {{- with .Description }}{{ $descr = . }} {{- else }}{{ with .Summary }}{{ $descr = (. | plainify | chomp | truncate 200) }} {{- else }}{{ $descr = site.Params.description }}{{ end }}{{ end }} {{- /* Taxonomy terms have no source file to carry a description, so build one rather than let every tag page repeat the site blurb verbatim — duplicate descriptions across pages are exactly what a SERP audit flags. */}} {{- if and .IsNode .Data.Term }} {{- /* Sized so the longest existing tag still lands inside the 120-160 window a SERP snippet rewards. A much longer tag would overshoot — worth a glance if you ever add one. */}} {{- $descr = printf "Posts tagged #%s — writing by %s on React, web performance, and the stubborn problems underneath ordinary frontend work." .Data.Term site.Params.author }} {{- end }} {{- /* Fall back to the search description, clipped to what a card actually shows. `truncate` breaks on a word, so the auto case still reads as a sentence; a post that wants a hand-written one sets ogDescription. */}} {{- $ogdescr := .Params.ogDescription | default ($descr | truncate 120) }} {{- /* One card for the whole site — the unfurl prints the title beside it, so the image is the mark, not the headline. A post can still opt out with `image:` in its front matter. */}} {{- $image := .Params.image | default "/blog/og/default.png" }} {{- /* A post's is the headline alone. Google truncates around 60 characters and appends the site name itself, so spending 24 of them on a suffix only buys a cut-off headline. List pages, whose titles are one word, keep a short brand tag. */}} {{- $brand := site.Params.brand | default site.Title }} <title>{{ if .IsHome }}{{ site.Title }}{{ else if .IsPage }}{{ .Title }}{{ else }}{{ .Title }} · {{ $brand }}{{ end }} {{- /* OG wants language_TERRITORY (en_US), not the BCP-47 en-us Hugo carries. */}} {{- $loc := split site.Language.Locale "-" }} {{- if and .IsPage (not .Date.IsZero) }} {{- if ne .Lastmod .Date }} {{- end }} {{- range .Params.tags }} {{- end }} {{- end }} {{- /* site = the account the card is attributed to, creator = who wrote it. One person writes everything here, so both point at the same handle. */}} {{- with site.Params.twitter }} {{- $handle := printf "@%s" (strings.TrimPrefix "@" .) }} {{- end }} {{/* ------------------------------------------------------------------ JSON-LD. The OG tags above are for social unfurls; this is the same facts in the vocabulary Google reads, and it's what makes a result eligible for an article rich card with a byline and date rather than a plain blue link. Only the home page and posts emit it. List and tag pages have nothing to say here that the crawler can't already see, and a CollectionPage stub earns nothing. Built as a dict and handed to `jsonify` so the escaping is the marshaller's problem — titles here carry apostrophes and em dashes, and hand-written JSON in a template gets that wrong eventually. ------------------------------------------------------------------ */}} {{- /* Every profile that is the same person, for entity reconciliation. */}} {{- $sameAs := slice }} {{- range $k, $v := site.Params.social }}{{ $sameAs = $sameAs | append $v }}{{ end }} {{- with site.Params.twitter }} {{- $sameAs = $sameAs | append (printf "https://x.com/%s" (strings.TrimPrefix "@" .)) }} {{- end }} {{- $person := dict "@type" "Person" "name" site.Params.author "url" site.BaseURL }} {{- with $sameAs }}{{ $person = merge $person (dict "sameAs" .) }}{{ end }} {{- $ld := dict }} {{- if .IsPage }} {{- $ld = dict "@context" "https://schema.org" "@type" "BlogPosting" "headline" .Title "description" $descr "url" .Permalink "mainEntityOfPage" (dict "@type" "WebPage" "@id" .Permalink) "image" ($image | absURL) "author" $person "publisher" $person }} {{- if not .Date.IsZero }} {{- $ld = merge $ld (dict "datePublished" (.Date.Format "2006-01-02T15:04:05Z07:00")) }} {{- $ld = merge $ld (dict "dateModified" (.Lastmod.Format "2006-01-02T15:04:05Z07:00")) }} {{- end }} {{- with .Params.tags }}{{ $ld = merge $ld (dict "keywords" (delimit . ", ")) }}{{ end }} {{- with .WordCount }}{{ $ld = merge $ld (dict "wordCount" .) }}{{ end }} {{- else if .IsHome }} {{- $ld = dict "@context" "https://schema.org" "@type" "WebSite" "name" site.Title "description" $descr "url" site.BaseURL "author" $person "publisher" $person }} {{- end }} {{- with $ld }} {{- end }}
